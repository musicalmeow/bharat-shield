Backend Changes

New DELETE Endpoint:

// DELETE /api/messages/:messageId
router.delete('/:messageId', async (req, res) => {
  const { messageId } = req.params;
  const { userId } = req.body;
  
  // Validate user can only delete their own messages
  const message = await Message.findById(messageId);
  if (message.senderId.toString() !== userId) {
    return res.status(403).json({ message: 'You can only delete your own messages' });
  }
  
  // Permanently delete from database
  await Message.findByIdAndDelete(messageId);
  
  res.json({ success: true, message: 'Message deleted successfully' });
});

Frontend Changes

Enhanced Delete Function:

const deleteMessage = async (messageId) => {
  // Show deletion animation
  setMessages(prev => prev.map(msg => 
    msg.id === messageId ? { ...msg, deleting: true } : msg
  ));
  
  // Call server to permanently delete
  const response = await fetch(`${getApiBase()}/api/messages/${messageId}`, {
    method: 'DELETE',
    body: JSON.stringify({ userId: currentUser.userId })
  });
  
  // Remove from UI after animation
  setTimeout(() => {
    setMessages(prev => prev.filter(msg => msg.id !== messageId));
  }, 300);
  
  // Update localStorage
  const updatedMessages = currentMessages.filter(msg => msg.id !== messageId);
  localStorage.setItem(storageKey, JSON.stringify(updatedMessages));
};
